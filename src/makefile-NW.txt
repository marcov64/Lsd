# specify where are the sources of LSD
SRC_DIR=$(SRC)/
LIB_DIR=$(SRC_DIR)lib/
CLIB_DIR=$(SRC_DIR)clib/
INCLUDE=-I$(SRC) -I$(PATH_HEADER)

# no-window specific variables
SUFFIX_NW=NW
LIB_NW=$(LIB) -lpthread
CC_NW=$(CC_CROSS)g++
SWITCH_CC_NW=$(GLOBAL_CC) $(SWITCH_CC) -D_NW_ -O3 -g0
TARGET_NW=lsd$(SUFFIX_NW)

# handle platform-specific stuff
ifeq ($(OS),Windows_NT)
  OBJ_DIR=$(TMP)/$(TARGET_NW)/
  EXE_EXT=.exe
  DLL_EXT=.dll
else
  ifeq ($(shell uname -s),Darwin)
    OBJ_DIR=$(TMPDIR)/$(TARGET_NW)/
    EXE_EXT=
    DLL_EXT=.dylib
  else
    OBJ_DIR=$(TMPDIR)/$(TARGET_NW)/
    EXE_EXT=
    DLL_EXT=.so
  endif
endif

# check if DLL is to be created
DLL=$(findstring $(strip $(DLL_NW)),false False FALSE no No NO 0)

# OS commands
RM=@rm$(EXE_EXT) -f
MKDIR=@mkdir$(EXE_EXT) -p

# targets
.PHONY: all
all: $(TARGET_NW)$(EXE_EXT)

ifeq ($(DLL),)
# dynamic library
lib$(TARGET_NW)$(DLL_EXT): $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o -L$(PATH_LIB) $(LIB_NW) -shared -o lib$(TARGET_NW)$(DLL_EXT)

# No Window executable (small file)
$(TARGET_NW)$(EXE_EXT): $(OBJ_DIR)LSDnw.o lib$(TARGET_NW)$(DLL_EXT)
	$(CC_NW) $(SWITCH_CC_NW) $(SWITCH_CC_LNK) $(OBJ_DIR)LSDnw.o -L. -l$(TARGET_NW) -L$(PATH_LIB) $(LIB_NW) -o $(TARGET_NW)$(EXE_EXT)
else
# No Window executable (single file)
$(TARGET_NW)$(EXE_EXT): $(OBJ_DIR)LSDnw.o $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o
	$(CC_NW) $(SWITCH_CC_NW) $(SWITCH_CC_LNK) $(OBJ_DIR)LSDnw.o $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o -L$(PATH_LIB) $(LIB_NW) -o $(TARGET_NW)$(EXE_EXT)
endif

# executable/DLL modules
$(OBJ_DIR)$(FUN).o: $(FUN).cpp $(LIB_DIR)libLSD.h $(LIB_DIR)check.h $(SRC_DIR)fun_head.h $(FUN_EXTRA)
	$(MKDIR) $(@D)
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(FUN).cpp -o $(OBJ_DIR)$(FUN).o

$(OBJ_DIR)LSDnw.o: $(SRC_DIR)LSDnw.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)LSDnw.cpp -o $(OBJ_DIR)LSDnw.o

	$(MKDIR) $(@D)
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(LIB_DIR)libmain.cpp -o $(OBJ_DIR)libmain.o

	$(MKDIR) $(@D)

# remove compiled files
clean:
	$(RM) $(OBJ_DIR)* lib$(TARGET_NW)$(DLL_EXT) $(TARGET_NW)$(EXE_EXT)
