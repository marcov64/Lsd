# specify where are the sources of LSD
SRC_DIR=$(SRC)/
LIB_DIR=$(SRC_DIR)lib/

# location of headers
INCLUDE=-I$(SRC) -I$(PATH_HEADER)

# no-window specific variables
SUFFIX_NW=NW
LIB_NW=$(LIB) -lpthread
CC_NW=$(CC_CROSS)g++
SWITCH_CC_NW=$(GLOBAL_CC) $(SWITCH_CC) -D_NW_ -O3 -g0
TARGET_NW=lsd$(SUFFIX_NW)

# location of object files
ifeq ($(OS),Windows_NT)
	OBJ_DIR=$(TMP)/$(TARGET_NW)/
	EXT=.exe
else
	OBJ_DIR=$(TMPDIR)/$(TARGET_NW)/
	EXT=
endif

# OS commands to delete files and create directories
RM=@rm$(EXT) -f
MKDIR=@mkdir$(EXT) -p

# targets
all: $(TARGET_NW)$(EXT)

# LSD No Window executable
$(TARGET_NW)$(EXT): $(OBJ_DIR)lsdnw.o $(OBJ_DIR)libmain.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN)$(SUFFIX_NW).o
	$(CC_NW) $(SWITCH_CC_NW) $(SWITCH_CC_LNK) $(OBJ_DIR)lsdnw.o $(OBJ_DIR)libmain.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN)$(SUFFIX_NW).o -L$(PATH_LIB) $(LIB_NW) -o $(TARGET_NW)$(EXT)

# executable modules
$(OBJ_DIR)$(FUN)$(SUFFIX_NW).o: $(FUN).cpp $(SRC_DIR)libLSD.h $(SRC_DIR)check.h $(SRC_DIR)fun_head.h $(FUN_EXTRA)
	$(MKDIR) $(@D)
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(FUN).cpp -o $(OBJ_DIR)$(FUN)$(SUFFIX_NW).o
