# specify where are the sources of LSD
SRC_DIR=$(SRC)/

# location of headers
INCLUDE=-I$(SRC) -I$(PATH_HEADER)

# common dependencies (all source files)
COM_DEP=$(SRC_DIR)decl.h $(SRC_DIR)common.h

# no-window specific variables
SUFFIX_NW=NW
LIB_NW=$(LIB) -lpthread
CC_NW=$(CC_CROSS)g++
SWITCH_CC_NW=$(GLOBAL_CC) $(SWITCH_CC) -D_NW_ -O3 -g0
TARGET_NW=lsd$(SUFFIX_NW)

# OS command to delete files
RM=rm -f

# link executable
$(TARGET_NW): $(FUN)$(SUFFIX_NW).o $(SRC_DIR)common.o $(SRC_DIR)lsd.o $(SRC_DIR)lsdmain.o $(SRC_DIR)filelib.o $(SRC_DIR)nets.o $(SRC_DIR)object.o $(SRC_DIR)util.o $(SRC_DIR)variab.o $(SRC_DIR)pugixml.o
	$(CC_NW) $(SWITCH_CC_NW) $(FUN)$(SUFFIX_NW).o $(SRC_DIR)common.o $(SRC_DIR)lsd.o $(SRC_DIR)lsdmain.o $(SRC_DIR)filelib.o $(SRC_DIR)nets.o $(SRC_DIR)object.o $(SRC_DIR)util.o $(SRC_DIR)variab.o $(SRC_DIR)pugixml.o $(SWITCH_CC_LNK) -L$(PATH_LIB) $(LIB_NW) -o $(TARGET_NW)

# compile modules
$(FUN)$(SUFFIX_NW).o: $(FUN).cpp $(FUN_EXTRA) $(SRC_DIR)check.h $(SRC_DIR)fun_head.h $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(FUN).cpp -o $(FUN)$(SUFFIX_NW).o

$(SRC_DIR)common.o: $(SRC_DIR)common.cpp $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)common.cpp -o $(SRC_DIR)common.o

$(SRC_DIR)lsd.o: $(SRC_DIR)lsd.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)lsd.cpp -o $(SRC_DIR)lsd.o

$(SRC_DIR)lsdmain.o: $(SRC_DIR)lsdmain.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)lsdmain.cpp -o $(SRC_DIR)lsdmain.o

$(SRC_DIR)filelib.o: $(SRC_DIR)filelib.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)filelib.cpp -o $(SRC_DIR)filelib.o

$(SRC_DIR)nets.o: $(SRC_DIR)nets.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)nets.cpp -o $(SRC_DIR)nets.o

$(SRC_DIR)object.o: $(SRC_DIR)object.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)object.cpp -o $(SRC_DIR)object.o

$(SRC_DIR)util.o: $(SRC_DIR)util.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)util.cpp -o $(SRC_DIR)util.o

$(SRC_DIR)variab.o: $(SRC_DIR)variab.cpp $(SRC_DIR)decl.h $(SRC_DIR)common.h
	$(CC_NW) $(SWITCH_CC_NW) $(INCLUDE) -c $(SRC_DIR)variab.cpp -o $(SRC_DIR)variab.o

$(SRC_DIR)pugixml.o: $(SRC_DIR)pugixml/pugixml.cpp $(SRC_DIR)pugixml/pugixml.hpp $(SRC_DIR)pugixml/pugiconfig.hpp
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(INCLUDE) -c $(SRC_DIR)pugixml/pugixml.cpp -o $(SRC_DIR)pugixml.o

# remove compiled files
clean:
	$(RM) $(SRC_DIR)common.o $(SRC_DIR)lsd.o $(SRC_DIR)lsdmain.o $(SRC_DIR)filelib.o $(SRC_DIR)nets.o $(SRC_DIR)object.o $(SRC_DIR)util.o $(SRC_DIR)variab.o $(SRC_DIR)pugixml.o $(FUN)$(SUFFIX_NW).o $(TARGET_NW) $(TARGET_NW).exe
