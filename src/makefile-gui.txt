# LSD file locations
SRC_DIR=$(LSDROOT)/$(SRC)/
LIB_DIR=$(SRC_DIR)lib/
CLIB_DIR=$(SRC_DIR)clib/
INCLUDE=-I$(LSDROOT)/$(SRC) -I$(PATH_HEADER)

# set general dependencies
DEP_LIB=$(LIB_DIR)libLSD.h $(SRC_DIR)makefile-gui.txt
DEP_GUI=$(DEP_LIB) $(SRC_DIR)LSD.h

# check if LSD source must be recompiled
ifeq ($(findstring $(strip $(TARGET)),LSD lSD LsD LSd Lsd lSd lsD lsd),)
  PRECOMP=false
else
  TARGET=LSD
  PRECOMP=$(findstring $(strip $(PRECOMPILED)),false False FALSE no No NO 0)
endif

# handle platform-specific stuff
ifeq ($(OS),Windows_NT)
  PLAT=windows
  TMP_DIR=$(TMP)
  EXE_EXT=.exe
  DLL_EXT=.dll
  AR_DIR=$(LSDROOT)/gnu/x86_64-w64-mingw32/bin/
  MANIF=$(TMP_DIR)/$(TARGET)/LSD.exe.manifest.o
  SWITCH_CC_DLL+=-shared
  SWITCH_CC_LNK+=-Wl,--whole-archive
  INCLUDE+=-I$(PATH_TCLTK_HEADER)
else
  ifeq ($(shell uname -s),Darwin)
    PLAT=mac
    TMP_DIR=$(TMPDIR)
    PKG=$(TARGET).app/Contents/MacOS/
    DLL_EXT=.dylib
    MAC_FRMWRK=-framework IOKit -framework CoreFoundation
    GLOBAL_CC+=-std=c++17 -mmacosx-version-min=11.0
    SWITCH_CC_LNK+=-L$(TARGET).app/Contents/MacOS
    SWITCH_CC_LNK_DLL+=-dynamiclib -flat_namespace -Wl,-all_load
    INCLUDE+=-I$(PATH_TCL_HEADER) -I$(PATH_TK_HEADER)
    DEP_LIB+=$(TARGET).app/Contents/Info.plist
    FIX_LSD=touch -c $(TARGET).app && \
      install_name_tool -change `otool -L $(PKG)$(TARGET) | sed -n 's/\(^[^/]*\(\/[^(]*Tcl\) .*$$\)/\2/p'` @executable_path/../Frameworks/Tcl.framework/Tcl $(PKG)$(TARGET) && \
      install_name_tool -change `otool -L $(PKG)$(TARGET) | sed -n 's/\(^[^/]*\(\/[^(]*Tk\) .*$$\)/\2/p'` @executable_path/../Frameworks/Tk.framework/Tk $(PKG)$(TARGET)
    FIX_DLL=install_name_tool -id @executable_path/lib$(TARGET)$(DLL_EXT) $(PKG)lib$(TARGET)$(DLL_EXT)
  else
    PLAT=linux
    TMP_DIR=/tmp
    DLL_EXT=.so
    SWITCH_CC_DLL+=-shared -fPIC
    SWITCH_CC_LNK+=-Wl,-rpath '-Wl,$$ORIGIN' -Wl,--whole-archive
    INCLUDE+=-I$(PATH_TCLTK_HEADER)
  endif
endif

# additional OS-specific locations
OBJ_DIR=$(TMP_DIR)/$(TARGET)/
CUR_DIR:=$(notdir $(patsubst %/,%,$(dir $(abspath $(lastword $(MAKEFILE_LIST))))))/

# select static libraries to link
ifeq ($(PRECOMP),)
  LIB_GUI=$(SRC_DIR)LSDgui-$(PLAT).a
  LIB_BASE=$(SRC_DIR)LSDbase-$(PLAT).a
else
  LIB_GUI=$(OBJ_DIR)LSDgui-$(PLAT).a
  LIB_BASE=$(OBJ_DIR)LSDbase-$(PLAT).a
endif

# OS commands
AR=$(AR_DIR)ar$(EXE_EXT) -rcs
RM=@rm$(EXE_EXT) -rf
MKDIR=@mkdir$(EXE_EXT) -p

# targets
.PHONY: all clean

ifeq ($(PRECOMP),)
all: $(PKG)lib$(TARGET)$(DLL_EXT)
else
all: $(PKG)$(TARGET)$(EXE_EXT)
endif

# macOS package
$(TARGET).app/Contents/Info.plist: $(LSDROOT)/LSD.app/Contents/MacOS/LSD $(LSDROOT)/LSD.app/Contents/Frameworks/Tcl.framework/libtclstub8.6.a $(LSDROOT)/LSD.app/Contents/Frameworks/Tk.framework/libtkstub8.6.a
	$(RM) LSD_APP_DUMMY $(TARGET).app
	cp -Rf $(LSDROOT)/LSD.app LSD_APP_DUMMY
	mv -f LSD_APP_DUMMY $(TARGET).app
	sed -e 's/LSD/$(TARGET)/g' -i '' $(TARGET).app/Contents/Info.plist

# dynamic library
$(PKG)lib$(TARGET)$(DLL_EXT): $(OBJ_DIR)$(CUR_DIR)$(FUN).o $(LIB_BASE)
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_DLL) $(OBJ_DIR)$(CUR_DIR)$(FUN).o $(SWITCH_CC_LNK) $(SWITCH_CC_LNK_DLL) $(LIB_BASE) -L$(PATH_LIB) $(LIB) $(MAC_FRMWRK) -o $(PKG)lib$(TARGET)$(DLL_EXT)
	$(FIX_DLL)

$(OBJ_DIR)$(CUR_DIR)$(FUN).o: $(FUN).cpp $(DEP_LIB) $(LIB_DIR)check.h $(SRC_DIR)fun_head.h $(FUN_EXTRA)
	$(MKDIR) $(@D)
	$(CC) $(GLOBAL_CC) $(SWITCH_CC) $(SWITCH_CC_DLL) $(INCLUDE) -c $(FUN).cpp -o $(OBJ_DIR)$(CUR_DIR)$(FUN).o
