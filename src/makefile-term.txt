# specify where are the sources of LSD
SRC_DIR=$(SRC)/
LIB_DIR=$(SRC_DIR)lib/
CLIB_DIR=$(SRC_DIR)clib/
INCLUDE=-I$(SRC) -I$(SRC)/lib -I$(PATH_HEADER)

# terminal specific variables
TARGET_LOW=$(shell echo $(TARGET) | tr$(EXE_EXT) A-z a-z)
TARGET_TERM=$(TARGET_LOW)_term
LIB_TERM=$(LIB) -lpthread
CC_TERM=$(CC_CROSS)g++
SWITCH_CC_TERM=$(GLOBAL_CC) $(SWITCH_CC) -D_TERM_
SWITCH_CP=-+ -3 -Wextra

# legacy compatibility for LSD equation file(s)
ifeq ($(EQUATION),)
  EQUATION=$(FUN)
endif
ifeq ($(EXTRA),)
  EXTRA=$(FUN_EXTRA)
endif

# handle platform-specific stuff
ifeq ($(OS),Windows_NT)
  GNU_DIR=$(LSDROOT)/gnu/
  GNU_BIN=$(GNU_DIR)bin/
  OBJ_DIR=$(TMP)/$(TARGET_TERM)/
  EXE_EXT=.exe
  DLL_EXT=.dll
  SWITCH_CC_DLL+=-shared
  CP_TERM=cython$(EXE_EXT)
  PYT_VER=$(shell $(GNU_BIN)python-version $(GNU_BIN))
  PYT_INCLUDE+=-I$(GNU_DIR)include/python$(PYT_VER)
  PYT_LIB_PATH=-L$(GNU_BIN)
  PYT_CFG=$(GNU_BIN)python-config $(PYT_VER)
else
  ifeq ($(shell uname -s),Darwin)
    OBJ_DIR=$(TMPDIR)/$(TARGET_TERM)/
    EXE_EXT=
    DLL_EXT=.dylib
    MAC_FRMWRK=-framework IOKit -framework CoreFoundation
    GLOBAL_CC+=-std=c++17 -mmacosx-version-min=11.0
    SWITCH_CC_LNK_DLL+=-dynamiclib -flat_namespace -install_name @executable_path/lib$(TARGET_TERM)$(DLL_EXT)
    CP_TERM=/usr/local/opt/cython/bin/cython
    PYT_VER=$(shell python3 -c "import sys; print('.'.join( map( str, sys.version_info[ : 2 ] ) ) )")
    PYT_LIB_PATH=$(shell python3 -c "from distutils import sysconfig; print( sysconfig.get_config_var( 'LIBDIR' ) )")
    PYT_CFG=python$(PYT_VER)-config
  else
    OBJ_DIR=/tmp/$(TARGET_TERM)/
    EXE_EXT=
    DLL_EXT=.so
    SWITCH_CC_DLL+=-shared -fPIC
    SWITCH_CC_LNK+=-Wl,-rpath '-Wl,$$ORIGIN'
    CP_TERM=cython
    PYT_VER=$(shell python3 -c "import sys; print('.'.join( map( str, sys.version_info[ : 2 ] ) ) )")
    PYT_LIB_PATH=$(shell python3 -c "from distutils import sysconfig; print( sysconfig.get_config_var( 'LIBDIR' ) )")
    PYT_CFG=python$(PYT_VER)-config
  endif
endif

# OS commands
RM=@rm$(EXE_EXT) -f
MKDIR=@mkdir$(EXE_EXT) -p

# check if debug compilation
DBG=$(findstring $(strip $(DEBUG_TERM)),true True TRUE yes Yes YES 1)

ifeq ($(DBG),)
  SWITCH_CC_TERM+= -O3 -g0
else
  SWITCH_CC_TERM+= -O0 -ggdb3
  SWITCH_CP+=--gdb --gdb-outdir .
endif

# check if DLL is to be created
DLL=$(findstring $(strip $(DLL_TERM)),false False FALSE no No NO 0)

# check if Python API dll is to be created
PYT=$(findstring $(strip $(PYTHON_API)),true True TRUE yes Yes YES 1)

ifneq ($(PYT),)
  DLL=
  PYT_TARGET=$(TARGET_LOW)_api
  PYT_INCLUDE+=$(shell $(PYT_CFG) --includes)
  PYT_EXT=$(shell $(PYT_CFG) --extension-suffix)
  PYT_LIB=$(PYT_TARGET)$(PYT_EXT)
endif

# targets
.PHONY: all
all: $(TARGET_TERM)$(EXE_EXT) $(PYT_LIB)
lib: lib$(TARGET_TERM)$(DLL_EXT)
python: $(PYT_LIB)

ifeq ($(DLL),)
# dynamic library
lib$(TARGET_TERM)$(DLL_EXT): $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(EQUATION).o
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_DLL) $(SWITCH_CC_LNK) $(SWITCH_CC_LNK_DLL) $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(EQUATION).o -L$(PATH_LIB) $(LIB_TERM) $(MAC_FRMWRK) -o lib$(TARGET_TERM)$(DLL_EXT)

# terminal executable (small file)
$(TARGET_TERM)$(EXE_EXT): $(OBJ_DIR)LSDterm.o lib$(TARGET_TERM)$(DLL_EXT)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_LNK) $(OBJ_DIR)LSDterm.o -L. -l$(TARGET_TERM) -L$(PATH_LIB) $(LIB_TERM) -o $(TARGET_TERM)$(EXE_EXT)
else
# terminal executable (single file)
$(TARGET_TERM)$(EXE_EXT): $(OBJ_DIR)LSDterm.o $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(EQUATION).o
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_LNK) $(OBJ_DIR)LSDterm.o $(OBJ_DIR)libmain.o $(OBJ_DIR)assimlib.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(EQUATION).o -L$(PATH_LIB) $(LIB_TERM) $(MAC_FRMWRK) -o $(TARGET_TERM)$(EXE_EXT)
endif

# executable/DLL modules
$(OBJ_DIR)$(EQUATION).o: $(EQUATION).cpp $(LIB_DIR)libLSD.h $(LIB_DIR)check.h $(LIB_DIR)lsdequation.h $(EXTRA)
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(EQUATION).cpp -o $(OBJ_DIR)$(EQUATION).o

$(OBJ_DIR)LSDterm.o: $(SRC_DIR)LSDterm.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(SRC_DIR)LSDterm.cpp -o $(OBJ_DIR)LSDterm.o

$(OBJ_DIR)libmain.o: $(LIB_DIR)libmain.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)libmain.cpp -o $(OBJ_DIR)libmain.o

$(OBJ_DIR)assimlib.o: $(LIB_DIR)assimlib.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)assimlib.cpp -o $(OBJ_DIR)assimlib.o

$(OBJ_DIR)commonlib.o: $(LIB_DIR)commonlib.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)commonlib.cpp -o $(OBJ_DIR)commonlib.o

$(OBJ_DIR)description.o: $(LIB_DIR)description.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)description.cpp -o $(OBJ_DIR)description.o

$(OBJ_DIR)filelib.o: $(LIB_DIR)filelib.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)filelib.cpp -o $(OBJ_DIR)filelib.o

$(OBJ_DIR)latticelib.o: $(LIB_DIR)latticelib.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)latticelib.cpp -o $(OBJ_DIR)latticelib.o

$(OBJ_DIR)math.o: $(LIB_DIR)math.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)math.cpp -o $(OBJ_DIR)math.o

$(OBJ_DIR)nets.o: $(LIB_DIR)nets.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)nets.cpp -o $(OBJ_DIR)nets.o

$(OBJ_DIR)object.o: $(LIB_DIR)object.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)object.cpp -o $(OBJ_DIR)object.o

$(OBJ_DIR)parallel.o: $(LIB_DIR)parallel.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)parallel.cpp -o $(OBJ_DIR)parallel.o

$(OBJ_DIR)run.o: $(LIB_DIR)run.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)run.cpp -o $(OBJ_DIR)run.o

$(OBJ_DIR)stattables.o: $(LIB_DIR)stattables.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)stattables.cpp -o $(OBJ_DIR)stattables.o

$(OBJ_DIR)utillib.o: $(LIB_DIR)utillib.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)utillib.cpp -o $(OBJ_DIR)utillib.o

$(OBJ_DIR)variab.o: $(LIB_DIR)variab.cpp $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(LIB_DIR)variab.cpp -o $(OBJ_DIR)variab.o

$(OBJ_DIR)pugixml.o: $(CLIB_DIR)pugixml.cpp $(CLIB_DIR)pugixml.hpp $(CLIB_DIR)pugiconfig.hpp
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) -c $(CLIB_DIR)pugixml.cpp -o $(OBJ_DIR)pugixml.o

# Python API dynamic library
$(PYT_LIB): $(OBJ_DIR)$(PYT_TARGET).o lib$(TARGET_TERM)$(DLL_EXT)
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_DLL) $(SWITCH_CC_LNK) $(SWITCH_CC_LNK_DLL) $(OBJ_DIR)$(PYT_TARGET).o -L. -l$(TARGET_TERM) -L$(PATH_LIB) $(LIB_TERM) $(PYT_LIB_PATH) -lpython$(PYT_VER) -o $(PYT_LIB)

$(OBJ_DIR)$(PYT_TARGET).o: $(OBJ_DIR)$(PYT_TARGET).cpp
	$(MKDIR) $(@D)
	$(CC_TERM) $(SWITCH_CC_TERM) $(SWITCH_CC_DLL) $(INCLUDE) $(PYT_INCLUDE) -c $(OBJ_DIR)$(PYT_TARGET).cpp -o $(OBJ_DIR)$(PYT_TARGET).o

$(OBJ_DIR)$(PYT_TARGET).cpp: $(LIB_DIR)pythonAPI.pyx $(LIB_DIR)pythonAPI.pxd $(LIB_DIR)libLSD.h
	$(MKDIR) $(@D)
	$(CP_TERM) $(SWITCH_CP) $(LIB_DIR)pythonAPI.pyx --module-name $(PYT_TARGET) -o $(OBJ_DIR)$(PYT_TARGET).cpp

# remove compiled files
clean:
	$(RM) $(OBJ_DIR)* lib$(TARGET_TERM)$(DLL_EXT) $(TARGET_TERM)$(EXE_EXT) $(PYT_LIB)
