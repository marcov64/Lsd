# specify where are the sources of LSD
SRC_DIR=$(LSDROOT)/$(SRC)/
LIB_DIR=$(SRC_DIR)lib/

# location of tcl/tk and other headers
INCLUDE=-I$(LSDROOT)\$(SRC) -I$(PATH_TCLTK_HEADER) -I$(PATH_HEADER)

# location of object files
OBJ_DIR=$(TMP)/$(TARGET)/

# OS commands to delete files and create directories
RM=@rm.exe -f
MKDIR=@mkdir.exe -p

all: $(TARGET).exe

# LSD executable
$(TARGET).exe: lib$(TARGET).dll $(OBJ_DIR)lsd.o $(OBJ_DIR)lsdmain.o $(OBJ_DIR)analysis.o $(OBJ_DIR)browser.o $(OBJ_DIR)common.o $(OBJ_DIR)debug.o $(OBJ_DIR)draw.o $(OBJ_DIR)edit.o $(OBJ_DIR)edit_dat.o $(OBJ_DIR)file.o $(OBJ_DIR)interf.o $(OBJ_DIR)lattice.o $(OBJ_DIR)log.o $(OBJ_DIR)nolhtables.o $(OBJ_DIR)report.o $(OBJ_DIR)runtime.o $(OBJ_DIR)set_all.o $(OBJ_DIR)sensitivity.o $(OBJ_DIR)show_eq.o $(OBJ_DIR)util.o $(OBJ_DIR)lsd_manifest.o
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)lsd.o $(OBJ_DIR)lsdmain.o $(OBJ_DIR)analysis.o $(OBJ_DIR)browser.o $(OBJ_DIR)common.o $(OBJ_DIR)debug.o $(OBJ_DIR)draw.o $(OBJ_DIR)edit.o $(OBJ_DIR)edit_dat.o $(OBJ_DIR)file.o $(OBJ_DIR)interf.o $(OBJ_DIR)lattice.o $(OBJ_DIR)log.o $(OBJ_DIR)nolhtables.o $(OBJ_DIR)report.o $(OBJ_DIR)runtime.o $(OBJ_DIR)set_all.o $(OBJ_DIR)sensitivity.o $(OBJ_DIR)show_eq.o $(OBJ_DIR)util.o $(OBJ_DIR)lsd_manifest.o -L. -l$(TARGET) -L$(PATH_TCLTK_LIB) $(TCLTK_LIB) -L$(PATH_LIB) $(LIB) -o $(TARGET).exe

# dynamic link library
lib$(TARGET).dll: $(OBJ_DIR)libmain.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)libmain.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o $(OBJ_DIR)$(FUN).o -L$(PATH_LIB) $(LIB) -shared -o lib$(TARGET).dll
