# check if LSD source must be recompiled
ifeq ($(findstring $(strip $(TARGET)),LSD lSD LsD LSd Lsd lSd lsD lsd),)
  PRECOMP=false
else
  TARGET=LSD
  PRECOMP=$(findstring $(strip $(PRECOMPILED)),false False FALSE no No NO 0)
endif

# select static libraries to link
ifeq ($(PRECOMP),)
  LIB_GUI=$(SRC_DIR)LSDgui-windows.a
  LIB_BASE=$(SRC_DIR)LSDbase-windows.a
else
  LIB_GUI=$(OBJ_DIR)LSDgui-windows.a

endif

DEP_LIB=$(LIB_DIR)libLSD.h $(SRC_DIR)system_options.txt

# OS commands and definitions

DLL_EXT=.dll
AR_DIR=$(LSDROOT)/gnu/x86_64-w64-mingw32/bin/
AR=$(AR_DIR)ar$(EXE_EXT) -rcs
RM=@rm$(EXE_EXT) -f
MKDIR=@mkdir$(EXE_EXT) -p

# targets
.PHONY: all clean

ifeq ($(PRECOMP),)
all: lib$(TARGET)$(DLL_EXT)
else
all: $(TARGET)$(EXE_EXT)
endif

# dynamic library
lib$(TARGET)$(DLL_EXT): $(OBJ_DIR)$(FUN).o $(LIB_BASE)
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)$(FUN).o -Wl,--whole-archive $(LIB_BASE) -Wl,--no-whole-archive -L$(PATH_LIB) $(LIB) -shared -o lib$(TARGET)$(DLL_EXT)

$(OBJ_DIR)$(FUN).o: $(FUN).cpp $(DEP_LIB) $(LIB_DIR)check.h $(SRC_DIR)fun_head.h $(FUN_EXTRA) model_options.txt

	$(CC) $(GLOBAL_CC) $(SWITCH_CC) $(INCLUDE) -c $(FUN).cpp -o $(OBJ_DIR)$(FUN).o
