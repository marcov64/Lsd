# specify where are the sources of LSD and destination of temporary files
SRC_DIR=$(LSDROOT)/$(SRC)/
LIB_DIR=$(SRC_DIR)lib/
OBJ_DIR=$(TMP)/$(TARGET)/

# location of tcl/tk and other headers
INCLUDE=-I$(LSDROOT)/$(SRC) -I$(PATH_TCLTK_HEADER) -I$(PATH_HEADER)

# OS-specific commands and definitions
EXE_EXT=.exe
DLL_EXT=.dll
AR_DIR=$(LSDROOT)/gnu/x86_64-w64-mingw32/bin/
AR=$(AR_DIR)ar$(EXE_EXT) -rcs
RM=@rm$(EXE_EXT) -f
MKDIR=@mkdir$(EXE_EXT) -p

# targets
.PHONY: all libdll libgui libbase clean
all: $(TARGET)$(EXE_EXT)
libgui: $(OBJ_DIR)libLSDgui.a
libdll: lib$(TARGET)$(DLL_EXT)
libbase: $(OBJ_DIR)libLSDbase.a

# LSD executable
$(TARGET)$(EXE_EXT): $(OBJ_DIR)lsd_manifest.o $(OBJ_DIR)libLSDgui.a lib$(TARGET)$(DLL_EXT)
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)lsd_manifest.o -L. -l$(TARGET) -L$(OBJ_DIR) -lLSDgui -L$(PATH_TCLTK_LIB) $(TCLTK_LIB) -L$(PATH_LIB) $(LIB) -o $(TARGET)$(EXE_EXT)

# dynamic library
lib$(TARGET)$(DLL_EXT): $(OBJ_DIR)$(FUN).o $(OBJ_DIR)libLSDbase.a
	$(CC) $(GLOBAL_CC) $(SSWITCH_CC) $(SWITCH_CC_LNK) $(OBJ_DIR)$(FUN).o -L$(OBJ_DIR) -Wl,--whole-archive -lLSDbase -Wl,--no-whole-archive -L$(PATH_LIB) $(LIB) -shared -o lib$(TARGET)$(DLL_EXT)

# GUI static library
$(OBJ_DIR)libLSDgui.a: $(OBJ_DIR)lsd.o $(OBJ_DIR)lsdmain.o $(OBJ_DIR)analysis.o $(OBJ_DIR)browser.o $(OBJ_DIR)common.o $(OBJ_DIR)debug.o $(OBJ_DIR)draw.o $(OBJ_DIR)edit.o $(OBJ_DIR)edit_dat.o $(OBJ_DIR)file.o $(OBJ_DIR)interf.o $(OBJ_DIR)lattice.o $(OBJ_DIR)log.o $(OBJ_DIR)nolhtables.o $(OBJ_DIR)report.o $(OBJ_DIR)runtime.o $(OBJ_DIR)set_all.o $(OBJ_DIR)sensitivity.o $(OBJ_DIR)show_eq.o $(OBJ_DIR)util.o
	$(AR) $(OBJ_DIR)libLSDgui.a $(OBJ_DIR)lsd.o $(OBJ_DIR)lsdmain.o $(OBJ_DIR)analysis.o $(OBJ_DIR)browser.o $(OBJ_DIR)common.o $(OBJ_DIR)debug.o $(OBJ_DIR)draw.o $(OBJ_DIR)edit.o $(OBJ_DIR)edit_dat.o $(OBJ_DIR)file.o $(OBJ_DIR)interf.o $(OBJ_DIR)lattice.o $(OBJ_DIR)log.o $(OBJ_DIR)nolhtables.o $(OBJ_DIR)report.o $(OBJ_DIR)runtime.o $(OBJ_DIR)set_all.o $(OBJ_DIR)sensitivity.o $(OBJ_DIR)show_eq.o $(OBJ_DIR)util.o

	$(AR) $(OBJ_DIR)libLSDbase.a $(OBJ_DIR)libmain.o $(OBJ_DIR)commonlib.o $(OBJ_DIR)description.o $(OBJ_DIR)filelib.o $(OBJ_DIR)latticelib.o $(OBJ_DIR)math.o $(OBJ_DIR)nets.o $(OBJ_DIR)object.o $(OBJ_DIR)parallel.o $(OBJ_DIR)run.o $(OBJ_DIR)stattables.o $(OBJ_DIR)utillib.o $(OBJ_DIR)variab.o $(OBJ_DIR)pugixml.o
